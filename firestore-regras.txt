rules_version = '2';

// ===================================================================
//  REGRAS DO FIRESTORE — 2ª Corrida Everest
//  Fase 1D: painel exige login Google + admins/{uid} (ativo, role admin)
//  Projeto: site-corrida  (site-corrida-8e4df)
//
//  Cole em: Firebase → Firestore Database → aba "Regras" → PUBLICAR
// ===================================================================

service cloud.firestore {
  match /databases/{database}/documents {

    // DOIS TIPOS DE QUEM ACESSA:
    //   · visitante (anônimo): vê a vitrine, se inscreve, consulta a própria
    //     inscrição, consome cupom e registra eventos de funil. Nada mais.
    //   · organizador: logado com Google E com admins/{uid} ativo e role
    //     'admin'. É o único que confirma, cancela, mexe em estoque, preço,
    //     cupom e lê a lista completa.
    // A decisão passa SEMPRE por ehAdmin(). Não escreva `request.auth != null`
    // solto em outro lugar: estar logado NÃO é ser organizador.

    // Limita o tamanho de um campo de texto, se ele existir.
    function textoAte(campo, max) {
      return !(campo in request.resource.data)
             || request.resource.data[campo] == null
             || (request.resource.data[campo] is string
                 && request.resource.data[campo].size() <= max);
    }

    // Barra documento gigante (alguém tentando entupir o banco).
    // O site grava 20 campos pelo formulário e 26 pelo balcão;
    // 40 dá folga sem abrir a porteira.
    function tamanhoSensato() {
      return request.resource.data.keys().size() <= 40;
    }

    function camposDeInscricao() {
      return tamanhoSensato()
             && textoAte('nome', 120)
             && textoAte('email', 120)
             && textoAte('celular', 30)
             && textoAte('documento', 30)
             && textoAte('cpf', 30)
             && textoAte('docTipo', 10)
             && textoAte('cidade', 80)
             && textoAte('status', 20)
             && textoAte('camiseta', 6)
             && textoAte('protocolo', 30)
             && textoAte('pagamento', 20)
             && textoAte('genero', 30)
             && textoAte('origem', 20)
             && textoAte('motivoDesconto', 200)
             // Quem retirou o kit. Sem estes três limites, seriam campos de
             // texto ILIMITADO graváveis por qualquer visitante — o site não
             // tem login, e textoAte só protege o que está listado aqui.
             && textoAte('retiradoPorTipo', 10)
             && textoAte('retiradoPorNome', 120)
             && textoAte('retiradoPorCpf', 14);
    }

    // ---------------- QUEM É ORGANIZADOR ----------------
    // As MESMAS três condições que o painel confere em souAdmin() no
    // index.html. Se as duas divergirem, o painel abre e a gravação falha.
    // Custa 1 leitura de admins/{uid} por gravação do painel. Para o
    // visitante não custa nada: sem login, para no primeiro teste.
    function adminDoc() {
      return /databases/$(database)/documents/admins/$(request.auth.uid);
    }
    function ehAdmin() {
      return request.auth != null
          && exists(adminDoc())
          && get(adminDoc()).data.get('ativo', false) == true
          && get(adminDoc()).data.get('role', '') == 'admin';
    }

    // ---------------- INSCRIÇÃO PELO SITE (visitante) ----------------
    // A forma EXATA que gerarInscricao() grava. Sem esta checagem, um
    // visitante criaria inscrição já 'confirmado', com estoque baixado,
    // cortesia ou kit entregue — vaga sem pagar, sem passar pelo organizador.
    //
    // ATENÇÃO: campo novo em gerarInscricao() tem de entrar no hasOnly
    // abaixo NA MESMA PUBLICAÇÃO, senão TODA inscrição pelo site é recusada.
    // O teste SKILL-CORRIDA/testes/regras.js cruza as duas listas.
    function criacaoPublica() {
      let d = request.resource.data;
      let s = get(/databases/$(database)/documents/config/settings).data;
      return d.keys().hasOnly([
               'protocolo','status','distanciaId','distancia',
               'camiseta','camisas','totalCamisas',
               'nome','docTipo','documento','cpf','nascimento','genero',
               'celular','email','cidade',
               'pagamento','valor','evento','origem','criadoEm','aceitaWpp',
               'estoqueBaixado',
               'valorCheio','cupom','cupomPercentual','cupomDesconto','cupomConsumido'])
          && d.status == 'pendente'
          && d.origem == 'site'
          && d.estoqueBaixado == false
          && d.pagamento in ['PIX', 'Dinheiro', 'Cartão']
          && d.valor is number && d.valor >= 0
          // camisas: só tamanhos que existem, e o total bate com a soma —
          // senão {G:50} com totalCamisas:1 tiraria 50 G do estoque na
          // confirmação. E o total respeita o limite comercial do painel.
          && d.camisas is map
          && d.camisas.keys().hasOnly(['PP','P','M','G','GG'])
          && d.totalCamisas is number
          && d.totalCamisas >= 1
          && d.totalCamisas == d.camisas.get('PP',0) + d.camisas.get('P',0)
                             + d.camisas.get('M',0)  + d.camisas.get('G',0)
                             + d.camisas.get('GG',0)
          && d.totalCamisas <= s.get('maxCamisasPorDoc', 1)
          // Prazo com o relógio DO SERVIDOR — atrasar o relógio do celular
          // não adianta. O balcão não passa por aqui (vende após o prazo).
          && (!('vendasAte' in s) || s.vendasAte == null
              || request.time.toMillis() < s.vendasAte);
    }

    // ---------------- INSCRIÇÕES ----------------
    match /inscricoes/{id} {

      // Ler UMA inscrição pelo ID: só o painel (transações). Nenhum fluxo
      // público lê por ID.
      allow get: if ehAdmin();

      // Consulta pública e checagem de CPF duplicado usam where(...) com
      // limit(5) e limit(10). Lista sem limite = painel, backup, planilha.
      // LIMITE CONHECIDO: o limit não impede paginar com cursor — ver
      // SKILL-CORRIDA/AUDITORIA-RULES-1D.md §F.2.
      allow list: if ehAdmin()
                  || (request.query.limit != null && request.query.limit <= 10);

      // Visitante: só na forma do site e dentro do prazo.
      // Organizador: qualquer forma (balcão, cortesia, restaurar backup).
      allow create: if camposDeInscricao() && (ehAdmin() || criacaoPublica());

      // Confirmar, cancelar, reabrir, kit, troca de tamanho, cupom: painel.
      allow update: if ehAdmin() && camposDeInscricao();

      // PERMANENTE. Nenhum código apaga inscrição desde a remoção do
      // "Apagar tudo e recomeçar" (24/09/2026). Inscrição é histórico.
      allow delete: if false;
    }

    // ---------------- CONFIGURAÇÕES (preços, lotes, estoque) ----------------
    match /config/settings {

      // A vitrine precisa ler preços, estoque e prazo antes de tudo.
      allow read: if true;

      // Só o painel grava. O site público nunca escreve aqui.
      // A forma continua exigida: evita que uma gravação torta derrube a página.
      allow create, update: if ehAdmin()
                   && request.resource.data.keys().hasAll(['versao','camiseta','lotes'])
                   && request.resource.data.keys().size() <= 20;
      allow delete: if false;
    }

    // ---------------- AUDITORIA (contadores de uso) ----------------
    match /eventos/{id} {
      allow read:   if ehAdmin();                        // só a auditoria lê
      allow create: if request.resource.data.keys().hasOnly(['tipo','em'])
                    && textoAte('tipo', 40)
                    && request.resource.data.em is number;  // DB.evento grava {tipo, em}
      allow update: if false;                            // histórico não se reescreve
      allow delete: if false;                            // PERMANENTE
    }

    // ---------------- CUPONS (desconto percentual) ----------------
    //
    // Sem este bloco a coleção cai no catch-all abaixo e NENHUM cupom
    // funciona — nem o painel, nem a inscrição.
    //
    // Criar, editar e apagar cupom: só o organizador (ehAdmin()).
    // O CONTADOR `usados` continua tendo um caminho público, porque o
    // visitante consome o cupom ao se inscrever — mas SÓ PARA CIMA. Devolver
    // uso (cancelar, trocar ou remover cupom de uma inscrição) é do painel.
    // Isso fecha o "reciclar cupom esgotado" pelo console do navegador.
    //
    // LIMITE CONHECIDO: um visitante ainda consegue SOMAR usos sem se
    // inscrever (queimar o cupom). Esta Rule não enxerga a inscrição da
    // mesma transação — ver AUDITORIA-RULES-1D.md §F.5.
    match /cupons/{codigo} {

      function d()     { return request.resource.data; }
      function antes() { return resource.data; }
      function mudou() { return d().diff(antes()).affectedKeys(); }

      // Documento fora desta forma é recusado na criação.
      // `mostrarNoSite` e `lotesPermitidos` são opcionais: cupom antigo
      // (criado antes deles) continua válido sem precisar de migração.
      function formaDeCupom() {
        return d().keys().hasOnly(['ativo','tipo','valor','limite','usados','validade','formasPermitidas','criadoEm','mostrarNoSite','lotesPermitidos'])
            && d().keys().hasAll(['ativo','tipo','valor','limite','usados','formasPermitidas','criadoEm'])
            && d().ativo is bool
            && d().tipo == 'percentual'
            && d().valor  is number && d().valor  >  0 && d().valor  <= 100
            && d().limite is number && d().limite >= 1 && d().limite <= 2000
            && d().usados is number
            && d().criadoEm is number
            && (!('validade' in d()) || d().validade == null || d().validade is number)
            && d().formasPermitidas is list
            && d().formasPermitidas.size() >= 1
            && d().formasPermitidas.size() <= 2
            && (!('mostrarNoSite' in d()) || d().mostrarNoSite is bool)
            && (!('lotesPermitidos' in d()) || (d().lotesPermitidos is list && d().lotesPermitidos.size() <= 10));
      }

      // EDIÇÃO PELO PAINEL. Sem login, isto vale para qualquer visitante —
      // por isso as três travas abaixo estão AQUI, e não só no JavaScript:
      //   · `usados` e `criadoEm` têm de sair iguais como entraram
      //   · o percentual só pode DIMINUIR (impede transformar um cupom
      //     existente em desconto maior)
      //   · `limite` nunca fica abaixo do que já foi usado
      // O código do cupom é o ID do documento, então não há como alterá-lo.
      function edicaoDeCupom() {
        return mudou().hasOnly(['ativo','valor','limite','validade','formasPermitidas','mostrarNoSite','lotesPermitidos'])
            && d().usados   == antes().usados
            && d().criadoEm == antes().criadoEm
            && d().tipo     == antes().tipo
            && d().ativo  is bool
            && d().valor  is number && d().valor  >  0 && d().valor <= antes().valor
            && d().limite is number && d().limite >= 1 && d().limite <= 2000
            && d().limite >= antes().usados
            && (!('validade' in d()) || d().validade == null || d().validade is number)
            && d().formasPermitidas is list
            && d().formasPermitidas.size() >= 1
            && d().formasPermitidas.size() <= 2
            && (!('mostrarNoSite' in d()) || d().mostrarNoSite is bool)
            && (!('lotesPermitidos' in d()) || (d().lotesPermitidos is list && d().lotesPermitidos.size() <= 10));
      }

      // CONSUMO E DEVOLUÇÃO. O cupom conta CAMISAS, não inscrições: uma
      // inscrição com 5 camisas move o contador em 5, não em 1. Por isso o
      // passo deixou de ser ±1 em 18/09/2026.
      //
      // O QUE ESTA REGRA GARANTE: só `usados` muda; continua entre 0 e
      // `limite`; e o salto cabe num teto de sanidade.
      //
      // O QUE ELA NÃO GARANTE — e não tem como: que o salto corresponda a uma
      // inscrição real. Rules avaliam um documento por vez, e `get()` lê o
      // estado já commitado; a Rule de `cupons` NÃO enxerga a inscrição
      // gravada na MESMA transação. Sem login, a proteção aqui é de FORMA.
      //
      // O teto de 50 não impede um atacante: sem login ele repete a operação
      // quantas vezes quiser — como já era possível com o passo de ±1, que
      // limitava a VELOCIDADE, não a possibilidade. O teto existe para barrar
      // BUG NOSSO: um número absurdo escrito por engano não corrompe o
      // contador de um cupom em produção.
      //
      // SE `maxCamisasPorDoc` PASSAR DE 50, É ESTA LINHA QUE VAI BLOQUEAR.
      function passoDeContador() {
        return mudou().hasOnly(['usados'])
            && d().usados is number
            && d().usados >= 0
            && d().usados <= antes().limite
            && d().usados != antes().usados
            && d().usados >= antes().usados - 50
            && d().usados <= antes().usados + 50;
      }

      // O site precisa ler o cupom para calcular o desconto e montar a vitrine.
      // PENDENTE (Publicação 2): restringir a lista a mostrarNoSite == true —
      // hoje a vitrine baixa também os cupons ocultos. Exige mudar o código
      // da vitrine ANTES desta Rule, senão a vitrine quebra.
      allow read: if true;

      // Cupom nasce sempre zerado, e só pelo painel.
      allow create: if ehAdmin() && formaDeCupom() && d().usados == 0;

      // Visitante: o contador só SOBE (consumo ao se inscrever).
      // Organizador: contador para cima ou para baixo, ou edição do cupom —
      // nunca as duas coisas na mesma gravação.
      allow update: if (passoDeContador() && d().usados > antes().usados)
                    || (ehAdmin() && (passoDeContador() || edicaoDeCupom()));

      // Some só o que nunca foi usado, e só pelo painel: uma inscrição no
      // histórico não pode apontar para um cupom que deixou de existir.
      allow delete: if ehAdmin() && resource.data.usados == 0;
    }

    // ---------------- ADMINISTRADORES DO PAINEL ----------------
    // O painel lê admins/{uid} para saber se a conta Google logada é
    // administradora. Cada usuário logado lê SÓ o próprio documento:
    // ninguém lista a coleção (não dá para descobrir quem são os admins)
    // e ninguém escreve por aqui — admin novo é criado à mão no Console.
    //
    // Sem este bloco a leitura cai no "TUDO O MAIS: fechado" abaixo e o
    // painel nega acesso a todo mundo, inclusive ao organizador.
    match /admins/{uid} {
      allow get: if request.auth != null && request.auth.uid == uid;
    }

    // ---------------- TUDO O MAIS: fechado ----------------
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
